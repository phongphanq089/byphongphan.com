## Usage

### Classic iOS Lockscreen

```tsx
import { SlideToUnlock } from "@/components/slide-to-unlock"

export function Example() {
  return (
    <SlideToUnlock
      variant="unlock"
      onConfirm={() => console.log("Device unlocked")}
    />
  )
}
```

### iOS Slide to Power Off

```tsx
import { SlideToUnlock } from "@/components/slide-to-unlock"

export function PowerOffExample() {
  return (
    <SlideToUnlock
      variant="power"
      label="slide to power off"
      confirmingLabel="Powering down…"
      doneLabel="Off"
      onConfirm={async () => {
        await fetch("/api/shutdown", { method: "POST" })
      }}
    />
  )
}
```

### Slide to Pay / Confirm Action

```tsx
import { SlideToUnlock } from "@/components/slide-to-unlock"

export function PaymentExample() {
  return (
    <SlideToUnlock
      variant="confirm"
      label="slide to pay $48.00"
      confirmingLabel="Processing…"
      doneLabel="Paid"
      onConfirm={async () => {
        await processCheckout()
      }}
    />
  )
}
```

## Props

| Prop | Type | Default | Description |
| :--- | :--- | :--- | :--- |
| `variant` | `"unlock" \| "power" \| "confirm"` | `"unlock"` | Preset styling and default labels. `"unlock"` for lockscreen, `"power"` for shutdown, `"confirm"` for actions/payments. |
| `label` | `string` | `undefined` | Custom track label. Automatically defaults to `"slide to unlock"`, `"slide to power off"`, or `"slide to confirm"`. |
| `confirmingLabel` | `string` | `undefined` | Text shown while `onConfirm` async promise is pending. |
| `doneLabel` | `string` | `undefined` | Text displayed once the operation succeeds. |
| `icon` | `React.ReactNode` | `undefined` | Custom icon rendered inside the thumb. |
| `onConfirm` | `() => Promise<unknown> \| unknown` | `undefined` | Action callback triggered when slider passes commitment threshold. |
| `resetAfter` | `number \| null` | `2400` | Time in ms before rewinding back to the start. Pass `null` to remain in the done state. |
| `height` | `number` | `56` | Track height in pixels. |
| `thumbWidth` | `number` | `64` | Width of the squircle thumb button. |
| `disabled` | `boolean` | `false` | Disables drag interaction and dims the control. |
