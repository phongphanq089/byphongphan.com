## Usage

```tsx
import { SpotlightCard } from "@/shared/ui/core/spotlight-card"

export function Example() {
  return (
    <SpotlightCard enableTilt spotlightSize={300} className="p-6">
      <h3 className="font-semibold text-foreground">Interactive Card</h3>
      <p className="text-sm text-muted-foreground">
        Move your mouse over this card to see the spotlight and 3D tilt effect.
      </p>
    </SpotlightCard>
  )
}
```

## Props

| Prop | Type | Default | Description |
| :--- | :--- | :--- | :--- |
| `spotlightSize` | `number` | `280` | Diameter of the radial spotlight in pixels. |
| `enableTilt` | `boolean` | `false` | Enables 3D perspective tilt following cursor position. |
| `glowRgb` | `string` | `undefined` | Custom RGB string (e.g. `"220, 38, 38"`). Defaults to adaptive light/dark values. |
| `glowColor` | `"neutral" \| "primary"` | `"neutral"` | Preset color for the spotlight glow. |
| `contentClassName` | `string` | `undefined` | Additional classes for the inner content wrapper. |
